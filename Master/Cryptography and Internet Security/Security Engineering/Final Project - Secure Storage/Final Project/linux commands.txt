LITERAL API ENDPOINTS (copiar e colar)

# 1) Criar ficheiro de teste
echo "Conteudo de teste para AES-GCM" > segredo.txt

# Upload encriptado
#NAO ENCRIPTA ASSIM
curl -X POST -H "Authorization: Bearer $API_TOKEN" -F "file=@segredo.txt" http://127.0.0.1:8000/files/

# Listar ficheiros (owned + shared)
curl -s -H "Authorization: Bearer $API_TOKEN" http://127.0.0.1:8000/files/

# Download de um ficheiro especifico (ciphertext)
curl -H "Authorization: Bearer $API_TOKEN" -o downloaded.enc http://127.0.0.1:8000/files/<file_id>/download

# Atualizar ficheiro (nova versão)
curl -X PUT -H "Authorization: Bearer $API_TOKEN" -F "file=@new_version.txt" http://127.0.0.1:8000/files/<file_id>

# Apagar ficheiro + limpeza de metadata
curl -X DELETE -H "Authorization: Bearer $API_TOKEN" http://127.0.0.1:8000/files/<file_id>

# Ver permissões do ficheiro
curl -H "Authorization: Bearer $API_TOKEN" http://127.0.0.1:8000/files/<file_id>/permissions

# Alterar permissões do ficheiro
curl -X PUT -H "Authorization: Bearer $API_TOKEN" -H "Content-Type: application/json" -d '{"permissions":"read"}' http://127.0.0.1:8000/files/<file_id>/permissions

# Partilhar com outro utilizador
curl -X POST -H "Authorization: Bearer $API_TOKEN" -H "Content-Type: application/json" -d '{"target_user_id":"<target_user_id>"}' http://127.0.0.1:8000/files/<file_id>/share

# Criar link anónimo (TTL protegido)
curl -X POST -H "Authorization: Bearer $API_TOKEN" http://127.0.0.1:8000/files/<file_id>/anonymous-link

# Download via link anónimo
curl -L -o anon_download.bin http://127.0.0.1:8000/anon/<link_id>

# Health check (saber se o server esta vivo)
curl -s http://127.0.0.1:8000/

# Verificar token OIDC autenticado
curl -s -H "Authorization: Bearer $API_TOKEN" http://127.0.0.1:8000/users/me
